"use client";

import { useCallback, useEffect, useState } from "react";
import {
  Plus, Play, Eye, Trash2, X, Zap, Tag,
  Pencil, Database, MessageSquare as MsgIcon,
} from "lucide-react";
import { api } from "@/lib/api";
import { cn } from "@/lib/utils";
import StatusBadge from "@/components/ui/status-badge";


// â”€â”€â”€ Edit form state shape â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
interface ProductForm {
  id: string;
  name: string;
  keywords: string; // comma separated
  images: string[];
  videos?: string[];
  documents?: string[];
  price?: number;
  currency?: string;
  description?: string;
}

interface EditForm {
  project:     string;
  headerType:  "None" | "Image" | "Video" | "Document";
  body:        string;
  footer:      string;
  messageType: "Text" | "Image" | "Video" | "Document" | "Audio";
  greeting:    string;
  products:    ProductForm[];
}

function freshEditForm(): EditForm {
  return {
    project: "",
    headerType: "None",
    body: "",
    footer: "",
    messageType: "Text",
    greeting: "Hi! How can I help you today?",
    products: [],
  };
}


// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
// Helpers
// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
function safeParseConfig(raw: any): any {
  if (!raw) return {};
  if (typeof raw === "string") {
    try { return JSON.parse(raw); } catch { return {}; }
  }
  return raw;
}

// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
export default function ChatbotPage() {
  const [flows,    setFlows]    = useState<any[]>([]);
  const [activeId, setActiveId] = useState<number|null>(null);
  const [detail,   setDetail]   = useState<any>(null);
  const [newName,  setNewName]  = useState("");
  const [showNew,  setShowNew]  = useState(false);
  const [nodeForm, setNodeForm] = useState({ nodeType:"message", text:"" });
  const [preview,  setPreview]  = useState<any>(null);
  const [conn,     setConn]     = useState({ source:0, target:0 });

  // Edit panel
  const [showEdit, setShowEdit] = useState(false);
  const [editForm, setEditForm] = useState<EditForm>(freshEditForm());

  const load = useCallback(async () => {
    try { setFlows((await api.get<any[]>("/bots")) || []); } catch {}
  }, []);
  useEffect(() => { load(); }, [load]);

  const open = async (id: number) => {
    setActiveId(id);
    setDetail(null);
    setPreview(null);
    setShowEdit(false);
    setNodeForm({ nodeType:"message", text:"" });
    setConn({ source:0, target:0 });
    try { setDetail(await api.get<any>(`/bots/${id}`)); } catch (e: any) { alert(e.message); }
  };

  const create = async () => {
    if (!newName) return;
    try { const f = await api.post<any>("/bots", { name:newName }); setShowNew(false); setNewName(""); load(); open(f.id); }
    catch (e: any) { alert(e.message); }
  };

  const addNode = async () => {
    if (!activeId || !nodeForm.text) return;
    try {
      const cfg: any = { text:nodeForm.text };
      if (nodeForm.nodeType==="question") cfg.answers=[{text:nodeForm.text,targetNodeId:null}];
      await api.post(`/bots/${activeId}/nodes`, { nodeType:nodeForm.nodeType, config:cfg });
      open(activeId); setNodeForm({ nodeType:"message", text:"" });
    } catch (e: any) { alert(e.message); }
  };

  const addConn = async () => {
    if (!activeId || !conn.source || !conn.target) return;
    try { await api.post(`/bots/${activeId}/connections`, conn); open(activeId); setConn({ source:0, target:0 }); }
    catch (e: any) { alert(e.message); }
  };

  const publish = async () => {
    if (!activeId) return;
    try { const r = await api.post<any>(`/bots/${activeId}/publish`); alert(`Flow "${r.name}" is now live`); open(activeId); load(); }
    catch (e: any) { alert(e.message); }
  };

  const showPreview = async () => {
    if (!activeId) return;
    try { setPreview(await api.get<any>(`/bots/${activeId}/preview`)); }
    catch (e: any) { alert(e.message); }
  };

  const remove = async (id: number) => {
    if (!confirm("Delete this flow?")) return;
    try { await api.del(`/bots/${id}`); if (activeId===id) { setDetail(null); setActiveId(null); } load(); }
    catch (e: any) { alert(e.message); }
  };

  // Open the edit panel â€” pre-fill from existing node data & bot config if available
  const openEdit = (detailOverride?: any) => {
    const d = detailOverride ?? detail;
    const savedConfig = safeParseConfig(d?.config);
    const firstMsgNode = d?.nodes?.find((n: any) => n.nodeType === "message");
    const firstMsgConfig = safeParseConfig(firstMsgNode?.config);

    const rawProducts = savedConfig?.products || firstMsgConfig?.products;
    let loadedProducts: ProductForm[] = [];
    if (Array.isArray(rawProducts)) {
      loadedProducts = rawProducts.map((p: any) => ({
        id: p.id || `prod-${Math.random().toString(36).substring(2, 7)}`,
        name: p.name || "",
        keywords: Array.isArray(p.keywords) ? p.keywords.join(", ") : p.keywords || "",
        images: Array.isArray(p.images) ? p.images : [],
        videos: Array.isArray(p.videos) ? p.videos : [],
        documents: Array.isArray(p.documents) ? p.documents : [],
        price: p.price,
        currency: p.currency || "â‚¹",
        description: p.description || "",
      }));
    }

    const bodyText = firstMsgConfig?.text || savedConfig?.greeting || "Hi! How can I help you today?";

    setEditForm({
      project:     d?.name || "",
      headerType:  "None",
      body:        bodyText,
      footer:      "",
      messageType: "Text",
      greeting:    savedConfig?.greeting || bodyText,
      products:    loadedProducts,
    });
    setShowEdit(true);
  };



  // Save edit â€” saves config to flow and node (uses existing API)
  const saveEdit = async () => {
    if (!activeId) return;
    try {
      const formattedProducts = editForm.products.map(p => ({
        id:          p.id,
        name:        p.name.trim(),
        keywords:    p.keywords.split(",").map(k => k.trim()).filter(Boolean),
        images:      (p.images || []).filter(img => img && img.trim() !== ""),
        videos:      (p.videos || []).filter(v => v && v.trim() !== ""),
        documents:   (p.documents || []).filter(doc => doc && doc.trim() !== ""),
        price:       p.price,
        currency:    p.currency || "â‚¹",
        description: p.description?.trim() || "",
      }));

      const botConfig = {
        greeting: editForm.body.trim() || editForm.greeting || "Hi! How can I help you today?",
        products: formattedProducts,
      };

      // 1. Try to update bot flow name on backend (non-critical â€” skip if endpoint missing)
      try {
        await api.put(`/bots/${activeId}`, {
          name: editForm.project.trim() || detail?.name,
          config: botConfig,
        });
      } catch { /* endpoint may not exist â€” config saved via node below */ }

      // 2. Save config inside first message node (always works via existing API)
      const firstMsg = detail?.nodes?.find((n: any) => n.nodeType === "message");

      if (firstMsg) {
        await api.put(`/bots/${activeId}/nodes/${firstMsg.id}`, {
          config: { text: botConfig.greeting, ...botConfig },
        });
      } else {
        await api.post(`/bots/${activeId}/nodes`, {
          nodeType: "message",
          config:   { text: botConfig.greeting, ...botConfig },
        });
      }

      setShowEdit(false);
      await load();
      open(activeId);
    } catch (e: any) {
      alert(e.message || "Failed to save.");
    }
  };


  // Publish from edit panel
  const publishFromEdit = async () => {
    await saveEdit();
    await publish();
  };

  const nodes = detail?.nodes || [];

  return (
    <div className="flex gap-3" style={{ height:"calc(100vh - 128px)" }}>

      {/* â”€â”€ Flow list â”€â”€ */}
      <div className="w-[252px] shrink-0 flex flex-col card-raised overflow-hidden">
        <div className="p-3 flex items-center justify-between" style={{ borderBottom:"1px solid rgba(34,197,94,0.18)" }}>
          <h3 className="font-bold text-[13px]" style={{ color:"var(--t1)" }}>Bot flows</h3>
          <button className="btn-primary !px-2.5 !py-1.5 text-xs" onClick={() => setShowNew(!showNew)}>
            <Plus size={11}/> New
          </button>
        </div>

        {showNew && (
          <div className="p-3 flex gap-2" style={{ borderBottom:"1px solid rgba(255,255,255,0.35)" }}>
            <input className="input text-xs" placeholder="Flow name" value={newName}
              onChange={e => setNewName(e.target.value)} onKeyDown={e => e.key==="Enter" && create()} />
            <button className="btn-primary !px-2.5 shrink-0" onClick={create}>Add</button>
          </div>
        )}

        <div className="flex-1 overflow-y-auto">
          {flows.map(f => (
            <button key={f.id} onClick={() => open(f.id)}
              className="w-full flex items-center justify-between px-4 py-3 text-left transition-colors"
              style={{
                borderBottom:"1px solid rgba(220,252,231,0.35)",
                background: activeId===f.id ? "rgba(220,252,231,0.45)" : "transparent",
              }}
              onMouseEnter={e => { if (activeId!==f.id) e.currentTarget.style.background="rgba(220,252,231,0.25)"; }}
              onMouseLeave={e => { if (activeId!==f.id) e.currentTarget.style.background="transparent"; }}>
              <div className="min-w-0">
                <p className="text-[13px] font-semibold truncate" style={{ color:"var(--t1)" }}>{f.name}</p>
                <p className="text-[11px] mt-0.5" style={{ color:"var(--t3)" }}>{f._count?.nodes || 0} nodes</p>
              </div>
              <div className="flex items-center gap-1.5 shrink-0">
                {f.published && <span className="badge-green text-[9px]">live</span>}
                {/* Edit button */}
                <button
                  className="btn-icon"
                  title="Edit"
                  onClick={async e => {
                    e.stopPropagation();
                    if (activeId === f.id && detail) {
                      openEdit();
                    } else {
                      // Load the bot first, then open edit with the loaded detail
                      setActiveId(f.id);
                      setDetail(null);
                      setPreview(null);
                      setNodeForm({ nodeType:"message", text:"" });
                      setConn({ source:0, target:0 });
                      try {
                        const loaded = await api.get<any>(`/bots/${f.id}`);
                        setDetail(loaded);
                        openEdit(loaded);
                      } catch (e: any) { alert(e.message); }
                    }
                  }}
                >
                  <Pencil size={11}/>
                </button>
                <button className="btn-icon" style={{ color:"#dc2626" }}
                  onClick={e => { e.stopPropagation(); remove(f.id); }}>
                  <Trash2 size={12}/>
                </button>
              </div>
            </button>
          ))}
          {flows.length===0 && (
            <div className="empty-state !py-10">
              <div className="empty-icon"><Zap size={18}/></div>
              <p className="empty-desc">No flows yet.</p>
            </div>
          )}
        </div>
      </div>

      {/* â”€â”€ Builder / Edit panel â”€â”€ */}
      <div className="flex-1 flex flex-col min-w-0 card-raised overflow-hidden">
        {!detail ? (
          <div className="flex-1 flex flex-col items-center justify-center gap-3">
            <div className="empty-icon icon-box-xl"><Zap size={22}/></div>
            <p className="empty-title">Select or create a flow</p>
            <p className="empty-desc">Choose a bot flow to start building.</p>
          </div>
        ) : showEdit ? (

          /* â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•
             EDIT PANEL  (matches reference UI)
             â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â• */
          <div className="flex flex-col h-full">
            {/* Header */}
            <div className="px-5 py-3 flex items-center gap-3 shrink-0"
              style={{ borderBottom:"1px solid rgba(34,197,94,0.18)", background:"rgba(220,252,231,0.20)" }}>
              <h3 className="font-bold text-[13px]" style={{ color:"var(--t1)" }}>{detail.name}</h3>
              <StatusBadge status={detail.published ? "approved" : "draft"} label={detail.published ? "Published" : "Draft"} />
              <div className="flex-1"/>
              <button className="btn-secondary !py-1.5 text-xs" onClick={() => setShowEdit(false)}>
                <X size={11}/> Close Edit
              </button>
            </div>

            {/* Scrollable form body */}
            <div className="flex-1 overflow-y-auto p-5 space-y-4">

              {/* â”€â”€ Project Selection â”€â”€ */}
              <div className="card-raised p-5">
                <div className="flex items-center gap-2 mb-4">
                  <Database size={15} style={{ color:"var(--gp)" }}/>
                  <h4 className="font-bold text-[13px]" style={{ color:"var(--t1)" }}>Project Selection</h4>
                </div>
                <div>
                  <label className="label">Bot / Project Name <span style={{ color:"#dc2626" }}>*</span></label>
                  <input
                    className="input text-xs"
                    placeholder="Enter bot flow / project name..."
                    value={editForm.project}
                    onChange={e => setEditForm({...editForm, project:e.target.value})}
                  />
                  <p className="text-[11px] mt-1.5" style={{ color:"var(--t3)" }}>
                    Configure the name of this bot flow for identification and embedding
                  </p>
                </div>

              </div>

              {/* â”€â”€ First Message on Customer Response â”€â”€ */}
              <div className="card-raised p-5">
                <div className="flex items-center gap-2 mb-5">
                  <MsgIcon size={15} style={{ color:"var(--gp)" }}/>
                  <h4 className="font-bold text-[13px]" style={{ color:"var(--t1)" }}>First Message on Customer Response</h4>
                </div>

                {/* Header message type tabs */}
                <div className="mb-4">
                  <label className="label">
                    Header message <span style={{ color:"#dc2626" }}>*</span>
                  </label>
                  <div className="tab-row mt-1 !p-1 w-fit">
                    {(["None","Image","Video","Document"] as const).map(t => (
                      <button key={t}
                        className={cn("tab-item !px-4 !py-1.5 !text-[12px]", editForm.headerType===t && "active")}
                        onClick={() => setEditForm({...editForm, headerType:t})}>
                        {t}
                      </button>
                    ))}
                  </div>
                </div>

                {/* Body message */}
                <div className="mb-4">
                  <label className="label">
                    Body message <span style={{ color:"#dc2626" }}>*</span>
                  </label>
                  <textarea
                    className="input"
                    rows={4}
                    placeholder="Enter the first message your chatbot will sendâ€¦"
                    value={editForm.body}
                    onChange={e => setEditForm({...editForm, body:e.target.value})}
                  />
                </div>

                {/* Footer */}
                <div className="mb-4">
                  <label className="label">Footer</label>
                  <input className="input" placeholder="Enter footer"
                    value={editForm.footer}
                    onChange={e => setEditForm({...editForm, footer:e.target.value})} />
                </div>

                {/* Message type */}
                <div>
                  <label className="label">
                    Select message type <span style={{ color:"#dc2626" }}>*</span>
                  </label>
                  <select className="input" value={editForm.messageType}
                    onChange={e => setEditForm({...editForm, messageType:e.target.value as EditForm["messageType"]})}>
                    {["Text","Image","Video","Document","Audio"].map(t => (
                      <option key={t}>{t}</option>
                    ))}
                  </select>
                </div>
              </div>

              {/* â”€â”€ Custom Products & Keywords Section â”€â”€ */}
              <div className="card-raised p-5">
                <div className="flex items-center justify-between mb-4">
                  <div className="flex items-center gap-2">
                    <Tag size={15} style={{ color:"var(--gp)" }}/>
                    <div>
                      <h4 className="font-bold text-[13px]" style={{ color:"var(--t1)" }}>Bot Products & Keyword Gallery</h4>
                      <p className="text-[11px]" style={{ color:"var(--t3)" }}>
                        User inbox mein product keywords type karega (e.g. &quot;ring&quot;), toh yeh multiple images dikhayega.
                      </p>
                    </div>
                  </div>
                  <button
                    type="button"
                    className="btn-primary !py-1 !px-3 text-xs"
                    onClick={() => {
                      const newProd: ProductForm = {
                        id: `prod-${Date.now()}`,
                        name: "New Product",
                        keywords: "item, product",
                        images: [
                          "https://images.pexels.com/photos/265906/pexels-photo-265906.jpeg?w=400&h=400&fit=crop",
                        ],
                        price: 999,
                        currency: "â‚¹",
                      };
                      setEditForm({ ...editForm, products: [...editForm.products, newProd] });
                    }}
                  >
                    + Add Product
                  </button>
                </div>

                <div className="space-y-4">
                  {editForm.products.map((p, pIdx) => (
                    <div key={p.id || pIdx} className="card p-4 rounded-xl relative space-y-3" style={{ background:"rgba(255,255,255,0.4)" }}>
                      <div className="flex items-center justify-between border-b pb-2" style={{ borderColor:"rgba(0,0,0,0.05)" }}>
                        <span className="font-semibold text-xs text-emerald-800">Product #{pIdx + 1}</span>
                        <button
                          type="button"
                          className="btn-icon text-red-500 hover:text-red-700"
                          onClick={() => {
                            const updated = editForm.products.filter((_, idx) => idx !== pIdx);
                            setEditForm({ ...editForm, products: updated });
                          }}
                        >
                          <Trash2 size={13} />
                        </button>
                      </div>

                      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                        <div>
                          <label className="label">Product Name <span style={{ color:"#dc2626" }}>*</span></label>
                          <input
                            className="input text-xs"
                            placeholder="e.g. Ring, Saree, Mobile"
                            value={p.name}
                            onChange={e => {
                              const updated = [...editForm.products];
                              updated[pIdx].name = e.target.value;
                              setEditForm({ ...editForm, products: updated });
                            }}
                          />
                        </div>
                        <div>
                          <label className="label">Price (Optional)</label>
                          <input
                            type="number"
                            className="input text-xs"
                            placeholder="e.g. 2499"
                            value={p.price || ""}
                            onChange={e => {
                              const updated = [...editForm.products];
                              updated[pIdx].price = Number(e.target.value) || undefined;
                              setEditForm({ ...editForm, products: updated });
                            }}
                          />
                        </div>
                      </div>

                      <div>
                        <label className="label">Trigger Keywords (comma separated) <span style={{ color:"#dc2626" }}>*</span></label>
                        <input
                          className="input text-xs"
                          placeholder="e.g. ring, rings, solitaire, diamond ring"
                          value={p.keywords}
                          onChange={e => {
                            const updated = [...editForm.products];
                            updated[pIdx].keywords = e.target.value;
                            setEditForm({ ...editForm, products: updated });
                          }}
                        />
                        <p className="text-[10px] mt-1" style={{ color:"var(--t3)" }}>
                          Jab user inbox mein inme se koi keyword bolega, yeh product show hoga.
                        </p>
                      </div>

                      {/* Images list for this product */}
                      <div>
                        <div className="flex items-center justify-between mb-2">
                          <label className="label mb-0">Product Images (Different Angles/Views)</label>
                          <div className="flex items-center gap-2">
                            <label className="btn-secondary !py-1 !px-2 text-[11px] cursor-pointer">
                              + Upload File
                              <input
                                type="file"
                                accept="image/*"
                                className="hidden"
                                onChange={async (e) => {
                                  const file = e.target.files?.[0];
                                  if (!file) return;
                                  try {
                                    const fd = new FormData();
                                    fd.append("file", file);
                                    const uploaded = await api.upload<any>("/upload", fd);
                                    const imgUrl = uploaded?.url || uploaded?.path || String(uploaded);
                                    const updated = [...editForm.products];
                                    updated[pIdx].images.push(imgUrl);
                                    setEditForm({ ...editForm, products: updated });
                                  } catch (err: any) {
                                    alert(err.message || "Image upload failed");
                                  }
                                }}
                              />
                            </label>
                            <button
                              type="button"
                              className="btn-secondary !py-1 !px-2 text-[11px]"
                              onClick={() => {
                                const updated = [...editForm.products];
                                updated[pIdx].images.push("https://images.pexels.com/photos/265906/pexels-photo-265906.jpeg?w=400&h=400&fit=crop");
                                setEditForm({ ...editForm, products: updated });
                              }}
                            >
                              + Add URL
                            </button>
                          </div>
                        </div>

                        <div className="space-y-2">
                          {p.images.map((imgUrl, imgIdx) => (
                            <div key={imgIdx} className="flex items-center gap-2">
                              <img
                                src={imgUrl}
                                alt={`View ${imgIdx + 1}`}
                                className="w-9 h-9 rounded object-cover border shrink-0 bg-white"
                                onError={(e) => { (e.target as HTMLElement).style.display = "none"; }}
                              />
                              <input
                                className="input text-xs flex-1"
                                placeholder="Image URL (http://...)"
                                value={imgUrl}
                                onChange={e => {
                                  const updated = [...editForm.products];
